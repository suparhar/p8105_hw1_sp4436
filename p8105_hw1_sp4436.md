library(ggplot2)

p <- ggplot(early_january_weather,
            aes(x = hour, y = temp, color = humid)) +
  geom_point(alpha = 0.7) +
  labs(title = "Temperature over Time (colored by Humidity)",
       x = "Time (hourly)", 
       y = "Temperature (°F)",
       color = "Humidity") +
  scale_color_viridis_c(option = "plasma") +
  theme_minimal()

p
